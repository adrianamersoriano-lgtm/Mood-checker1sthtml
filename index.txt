  <script>
    // Simple keyword-based mood detection
    const happyWords = ['happy','joy','glad','good','great','awesome','excited','love','amazing','fantastic']
    const sadWords = ['sad','down','unhappy','depressed','bad','lonely','upset','angry','miserable']

    const input = document.getElementById('moodInput')
    const faceCard = document.getElementById('faceCard')
    const moodText = document.getElementById('moodText')

    function detectMood(text){
      if(!text || !text.trim()) return 'neutral'
      const t = text.toLowerCase()
      let happyScore = 0, sadScore = 0
      happyWords.forEach(w=>{ if(t.includes(w)) happyScore++ })
      sadWords.forEach(w=>{ if(t.includes(w)) sadScore++ })
      if(happyScore > sadScore && happyScore>0) return 'happy'
      if(sadScore > happyScore && sadScore>0) return 'sad'
      // if both detected or none, choose neutral unless one clearly higher
      return 'neutral'
    }

    function updateMood(mood){
      faceCard.classList.remove('happy','sad','neutral')
      faceCard.classList.add(mood)
      faceCard.setAttribute('aria-label', mood + ' face')
      moodText.textContent = mood.charAt(0).toUpperCase()+mood.slice(1)
    }

    input.addEventListener('input', e=>{
      const mood = detectMood(e.target.value)
      updateMood(mood)
    })

    document.getElementById('happyBtn').addEventListener('click', ()=>{
      updateMood('happy')
      input.value = 'I am happy :)'
    })
    document.getElementById('sadBtn').addEventListener('click', ()=>{
      updateMood('sad')
      input.value = 'Feeling sad'
    })
    document.getElementById('neutralBtn').addEventListener('click', ()=>{
      updateMood('neutral')
      input.value = ''
    })

    // accessibility: allow Enter to submit detection
    input.addEventListener('keydown', e=>{ if(e.key === 'Enter') { e.preventDefault(); updateMood(detectMood(input.value)) } })
  </script>
